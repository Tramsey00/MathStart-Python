import importlib.util
import json
import sys

assert importlib.util.find_spec('django') is None
from backend.infrastructure.http.runtime import application
app = application()
operations = len(app.state.implemented_routes)
assert operations == 32, operations
assert not any(name == 'django' or name.startswith('django.') for name in sys.modules)
print(json.dumps({'runtime_application_startup': 'PASS', 'operations': operations,
                  'django_installed': False, 'django_imported': False}))
